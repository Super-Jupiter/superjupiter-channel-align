cmake_minimum_required(VERSION 3.16)
project(SuperJupiterChannelAlignConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_CHANNEL_ALIGN_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_CHANNEL_ALIGN_SOURCE_DIR}" channel-align)
  if(TARGET superjupiter-channel-align-tests OR TARGET superjupiter-channel-align-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterChannelAlign 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-channel-align-consumer main.cpp)
target_link_libraries(superjupiter-channel-align-consumer PRIVATE SuperJupiter::ChannelAlign)
